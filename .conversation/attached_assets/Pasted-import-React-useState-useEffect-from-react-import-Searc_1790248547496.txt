const TestEngineView = ({ test, navigate }) => {
  const [currentQIndex, setCurrentQIndex] = useState(0);
  const [answers, setAnswers] = useState({});
  const questions = test?.type === 'TCS' ? TCS_QUESTIONS : MAANG_QUESTIONS;

  const submitTest = async () => {
    let score = 0;
    questions.forEach(q => { if (answers[q.id] === q.correct) score++; });
    
    try {
      await fetch('http://localhost:5000/api/test/result', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ testTitle: test.title, score, total: questions.length })
      });
    } catch(e) { console.warn("Backend API offline, displaying local results."); }

    navigate('test_result', { testTitle: test.title, score, total: questions.length, answers, questions });
  };

  const currentQ = questions[currentQIndex];

  return (
    <div className="flex flex-col h-full app-bg">
      <div className="h-20 app-hl flex justify-between px-8 items-center shrink-0">
        <div className="font-black uppercase tracking-widest text-lg">{test?.title}</div>
        <button onClick={submitTest} className="app-bg app-border border-2 px-8 py-3 text-sm font-black uppercase hover:opacity-80">Submit & End</button>
      </div>
      <div className="flex-1 p-12 overflow-y-auto max-w-4xl mx-auto w-full">
        <div className="text-xs font-bold uppercase tracking-widest app-text-alt mb-4">Question {currentQIndex + 1} of {questions.length}</div>
        <h2 className="text-2xl font-black uppercase mb-12 border-b-4 app-border pb-6">{currentQ.text}</h2>
        <div className="space-y-4 mb-16">
          {currentQ.options.map((opt, idx) => (
            <label key={idx} className={`block p-6 border-4 cursor-pointer transition-all ${answers[currentQ.id] === idx ? 'app-border app-hl' : 'border-transparent app-bg-alt'}`} onClick={() => setAnswers({...answers, [currentQ.id]: idx})}>
              <span className="text-lg font-bold">{opt}</span>
            </label>
          ))}
        </div>
        <div className="flex justify-between border-t-2 app-border-t pt-8">
          <button onClick={() => setCurrentQIndex(Math.max(0, currentQIndex-1))} disabled={currentQIndex===0} className="px-8 py-4 border-2 app-border font-black uppercase disabled:opacity-20">Previous</button>
          <button onClick={() => setCurrentQIndex(Math.min(questions.length-1, currentQIndex+1))} disabled={currentQIndex===questions.length-1} className="px-8 py-4 app-hl font-black uppercase disabled:opacity-20">Next Node</button>
        </div>
      </div>
    </div>
  );
};
